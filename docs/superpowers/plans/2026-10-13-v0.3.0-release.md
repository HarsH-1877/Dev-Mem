# Dev-Memo v0.3.0 Release Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make `main` safe to publish as `dev-memo@0.3.0`. Fix the four release blockers found in the 2026-10-13 audit and add a `doctor` command that would have caught them.

**Architecture:** The hook path resolver currently lives as an inline string inside `cli/index.ts`. It moves into its own module, `cli/hook-resolver.ts`, so it can be tested by writing it to disk and running it. Generated hooks get a version marker. `status` uses that marker to spot hooks in the old format, and the new `doctor` command checks the whole install.

**Tech Stack:** TypeScript (ESM, `.js` import suffixes), Node ≥ 22, Vitest, native `fetch`.

## Global Constraints

- Node engine: `>=22` (unchanged). CI matrix: `[22.x]`.
- No new npm runtime dependencies.
- Generated hook scripts must never contain a path tied to one machine.
- Hooks must never throw or exit non-zero because Dev-Mem failed. A failure means "skip silently".
- Worst-case hook latency when the resolver has no cache: **< 4 s**. Typical latency: **< 150 ms**.
- `.dev-mem/` stays git-ignored. The resolver cache lives at `.dev-mem/install-path.txt`.
- All existing tests must stay green (`npm test`, currently 104 passing).
- Commit style: Conventional Commits (`fix:`, `feat:`, `chore:`, `docs:`).

- Environment variables the generated code reads (for tests and power users):
  - `DEV_MEM_DIST_DIR`: explicit override, checked first.
  - `DEV_MEM_GLOBAL_ROOTS`: extra `node_modules` folders, separated by `path.delimiter`, checked before the built-in candidates.
  - `DEV_MEM_SKIP_NPM=1`: never spawn `npm` (used in tests).

Resolution order (first valid match wins; "valid" means `<dir>/adapters` exists):
1. `DEV_MEM_DIST_DIR`
2. `<cwd>/.dev-mem/install-path.txt`
3. `import.meta.resolve("dev-memo/package.json")` + `/../dist`
4. `<root>/dev-memo/dist`, where `root` is each entry of `DEV_MEM_GLOBAL_ROOTS`, then `%APPDATA%/npm/node_modules`, `dirname(execPath)/node_modules`, `dirname(execPath)/../lib/node_modules`, `/usr/local/lib/node_modules`, `/opt/homebrew/lib/node_modules`
5. `npm root -g` with a **3000 ms timeout** (skipped when `DEV_MEM_SKIP_NPM=1`)

```ts
/**
 * Builds the ESM source that installed hook scripts use to locate the
 * dev-memo dist directory at runtime, so no machine-specific path is ever
 * written into a committed hook file.
 */
export const HOOK_MARKER = "// dev-mem-hook-v2";

export function buildResolverSource(): string {
  return `${HOOK_MARKER}
import { readFileSync, existsSync, writeFileSync } from "node:fs";
import { join, dirname, delimiter } from "node:path";
import { execFileSync } from "node:child_process";
import { pathToFileURL, fileURLToPath } from "node:url";

const __devMemCache = join(process.cwd(), ".dev-mem", "install-path.txt");
const __devMemValid = (d) => { try { return !!d && existsSync(join(d, "adapters")); } catch { return false; } };
const __devMemSave = (d) => { try { if (existsSync(dirname(__devMemCache))) writeFileSync(__devMemCache, d, "utf8"); } catch {} };

let distDir = "";

if (__devMemValid(process.env.DEV_MEM_DIST_DIR)) distDir = process.env.DEV_MEM_DIST_DIR;

if (!distDir) {
  try { const c = readFileSync(__devMemCache, "utf8").trim(); if (__devMemValid(c)) distDir = c; } catch {}
}

if (!distDir) {
  try {
    const p = join(fileURLToPath(import.meta.resolve("dev-memo/package.json")), "..", "dist");
    if (__devMemValid(p)) { distDir = p; __devMemSave(p); }
  } catch {}
}

if (!distDir) {
  const exeDir = dirname(process.execPath);
  const roots = [
    ...(process.env.DEV_MEM_GLOBAL_ROOTS ? process.env.DEV_MEM_GLOBAL_ROOTS.split(delimiter) : []),
    process.env.APPDATA ? join(process.env.APPDATA, "npm", "node_modules") : "",
    join(exeDir, "node_modules"),
    join(exeDir, "..", "lib", "node_modules"),
    "/usr/local/lib/node_modules",
    "/opt/homebrew/lib/node_modules",
  ];
  for (const r of roots) {
    const p = r ? join(r, "dev-memo", "dist") : "";
    if (__devMemValid(p)) { distDir = p; __devMemSave(p); break; }
  }
}

if (!distDir && process.env.DEV_MEM_SKIP_NPM !== "1") {
  try {
    const root = execFileSync(process.platform === "win32" ? "npm.cmd" : "npm", ["root", "-g"], {
      encoding: "utf8", timeout: 3000, stdio: ["ignore", "pipe", "ignore"], shell: process.platform === "win32",
    }).trim();
    const p = join(root, "dev-memo", "dist");
    if (__devMemValid(p)) { distDir = p; __devMemSave(p); }
  } catch {}
}
`;
}
```

import { readdirSync } from "node:fs";
import { HOOK_MARKER } from "./hook-resolver.js";

export function findStaleHooks(cwd: string): string[] {
  const locations: Array<[string, RegExp]> = [
    [".claude/hooks", /^dev-mem-.*\.js$/],
    [".codex/hooks", /^dev-mem-.*\.js$/],
    [".cursor/hooks", /^dev-mem-.*\.js$/],
    [".opencode/plugins", /^dev-mem.*\.js$/],
  ];
  const stale: string[] = [];
  for (const [rel, pattern] of locations) {
    const dir = join(cwd, rel);
    if (!existsSync(dir)) continue;
    for (const name of readdirSync(dir)) {
      if (!pattern.test(name)) continue;
      try {
        if (!readFileSync(join(dir, name), "utf8").includes(HOOK_MARKER)) stale.push(`${rel}/${name}`);
      } catch {}
    }
  }
  return stale;
}
```

  const fail = (m: string) => { ok = false; lines.push(`✗ ${m}`); };

  try {
    execFileSync("git", ["rev-parse", "--show-toplevel"], { cwd, stdio: "ignore" });
    lines.push("✓ Git repository");
  } catch { fail("Not a git repository"); }

  const agents = [
    ["claude-code", ".claude/hooks"], ["codex", ".codex/hooks"],
    ["cursor", ".cursor/hooks"], ["opencode", ".opencode/plugins"],
  ].filter(([, rel]) => existsSync(join(cwd, rel)) &&
    readdirSync(join(cwd, rel)).some((f) => f.startsWith("dev-mem"))).map(([a]) => a);
  lines.push(agents.length ? `✓ Hooks: ${agents.join(", ")}` : "! No agent hooks installed — run dev-memo install");

  const stale = findStaleHooks(cwd);
  lines.push(stale.length ? `! ${stale.length} stale hook file(s) — run dev-memo install` : "✓ Hooks up to date");

  const probeDir = mkdtempSync(join(tmpdir(), "dm-doctor-"));
  const probe = join(probeDir, "probe.mjs");
  writeFileSync(probe, `${buildResolverSource()}\nprocess.stdout.write(distDir);\n`);
  const res = spawnSync(process.execPath, [probe], { cwd, encoding: "utf8", timeout: 5000 });
  rmSync(probeDir, { recursive: true, force: true });
  const found = (res.stdout || "").trim();
  if (found) lines.push(`✓ dev-memo found at ${found}`); else fail("Hooks cannot locate dev-memo");

  const provider = resolveProviderForProject(cwd);
  if (!provider) {
    fail("No LLM provider configured");
  } else {
    lines.push(`✓ LLM provider: ${provider.name}`);
    if (opts.live) {
      try {
        await Promise.race([
          provider.complete("Reply with OK.", "OK", 5),
          new Promise((_, rej) => setTimeout(() => rej(new Error("timeout after 15s")), 15000)),
        ]);
        lines.push("✓ LLM reachable");
      } catch (e) { fail(`LLM call failed: ${(e as Error).message}`); }
    }
  }
  return { ok, lines };
}
```
