# Dev-Mem v0.4.0 Competitive Upgrade Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Upgrade Dev-Mem from a basic session-end memory logger into a high-precision, token-frugal, team-syncable engineering memory platform with prompt-time injection, secret scrubbing, smart compaction, and a zero-daemon stdio MCP server.

**Architecture:** The upgrade is organized into 5 sequential phases (8 self-contained tasks). Every phase preserves Dev-Mem's core invariants: zero background daemons, zero new runtime npm dependencies, and a strict ceiling of at most 1 LLM call per session.

**Tech Stack:** TypeScript (ESM, `.js` import suffixes), Node.js >= 22, `better-sqlite3` (WAL mode), Vitest, JSON-RPC 2.0 over stdio (MCP).

## Global Constraints

- Node engine: `>=22`. CI matrix: `[22.x]`.
- Zero new npm runtime dependencies (`dependencies` in `package.json` must remain only `better-sqlite3`).
- Strict 1-LLM-call-per-session ceiling: retrieval, stemming, compaction, secret scrubbing, MCP tools, and git sync must make **zero** LLM API calls.
- Hook safety: no hook or CLI background step may ever throw an unhandled error or exit non-zero to block the host coding agent.
- All existing tests must remain green (`npm test`).
- Commit style: Conventional Commits (`feat:`, `fix:`, `chore:`, `docs:`).

```ts
/**
 * Lightweight, zero-dependency English suffix stemmer for code & prose tokens.
 * Normalises inflections (authentication/authenticating/authenticated → authent)
 * without external NLP libraries.
 */
export function stemToken(word: string): string {
  let w = word.toLowerCase();
  if (w.length <= 4) return w;

  // Step 1: plurals and -ied / -ies
  if (w.endsWith("ies") && w.length > 4) {
    w = w.slice(0, -3) + "y";
  } else if (w.endsWith("ied") && w.length > 4) {
    w = w.slice(0, -3) + "y";
  } else if (w.endsWith("es") && (w.endsWith("sses") || w.endsWith("ches") || w.endsWith("shes") || w.endsWith("xes"))) {
    w = w.slice(0, -2);
  } else if (w.endsWith("s") && !w.endsWith("ss") && !w.endsWith("us") && !w.endsWith("is")) {
    w = w.slice(0, -1);
  }

  // Step 2: derivational & inflectional suffixes (longest match first)
  const suffixes = [
    "ications", "ication", "izations", "ization", "ational", "fulness",
    "ousness", "iveness", "ements", "ement", "ating", "ation", "ances",
    "ance", "ences", "ence", "ments", "ment", "ings", "ing", "able",
    "ible", "ated", "ates", "ate", "ized", "izes", "ize", "full",
    "less", "ness", "ers", "er", "ed", "ly",
  ];
  for (const suffix of suffixes) {
    if (w.endsWith(suffix) && w.length - suffix.length >= 4) {
      w = w.slice(0, -suffix.length);
      break;
    }
  }

  // Step 3: trailing silent 'e' or doubled consonant cleanup
  if (w.endsWith("e") && w.length > 4) {
    w = w.slice(0, -1);
  }
  if (w.length > 4 && /([bcdfghjklmnpqrstvwxz])\1$/.test(w)) {
    w = w.slice(0, -1);
  }
  return w;
}
```

Also add `promoteSurvivingNodes` to `core/consistency/index.ts`:

```ts
/**
 * Promotes `observed` nodes to `active` when a subsequent session touches or
 * validates their cited files and the files still exist on disk.
 */
export function promoteSurvivingNodes(
  projectRoot: string,
  touchedFiles: string[],
  excludeNodeIds?: string[],
  dbPath?: string,
): string[] {
  const promoted: string[] = [];
  if (touchedFiles.length === 0) return promoted;

  const excludeSet = new Set(excludeNodeIds || []);
  const norm = (p: string) => p.replace(/\\/g, "/").replace(/^\.\//, "");
  const touchedSet = new Set(touchedFiles.map(norm));

  const store = new GraphStore({ projectRoot, dbPath, ephemeral: dbPath !== undefined });
  try {
    const allNodes = store.queryAllNodes();
    for (const node of allNodes) {
      if (excludeSet.has(node.id)) continue;
      if (node.lifecycle_state !== "observed") continue;

      const overlapsTouched = node.evidence.files.some((f) => touchedSet.has(norm(f)));
      if (!overlapsTouched) continue;

      const anyFileExists = node.evidence.files.some((f) => existsSync(join(projectRoot, f)));
      if (!anyFileExists) continue;

      try {
        store.transitionLifecycleState(node.id, "active");
        promoted.push(node.id);
      } catch {
        // Ignore invalid transition
      }
    }
  } catch {
    // Ignore store read errors
  } finally {
    store.close();
  }
  return promoted;
}
```

```ts
  // OpenAI / Anthropic / similar sk- keys
  /\bsk-[a-zA-Z0-9_-]{16,}\b/g,
  // Google / Gemini API keys
  /\bAIza[0-9A-Za-z_-]{30,}\b/g,

  /\b(?:gh[pousr]_[A-Za-z0-9_]{20,}|github_pat_[A-Za-z0-9_]{20,})\b/g,

  /(Bearer\s+)[A-Za-z0-9._~+/=-]{16,}/gi,
  // Common ENV assignments of secrets/passwords/tokens
  /((?:API_KEY|SECRET|TOKEN|PASSWORD|PRIVATE_KEY|CLIENT_SECRET)\s*[=:]\s*['"]?)[^\s'"]{8,}/gi,
];

/**
 * Redacts <private>...</private> blocks and common API keys/tokens from text.
 */
export function sanitizeText(input: string): string {
  if (!input) return input;
  let out = input.replace(/<private>[\s\S]*?<\/private>/gi, "[REDACTED_PRIVATE]");
  for (const pattern of SECRET_PATTERNS) {
    out = out.replace(pattern, (_match, prefix) =>
      typeof prefix === "string" ? `${prefix}[REDACTED_SECRET]` : "[REDACTED_SECRET]"
    );
  }
  return out;
}

/**
 * Returns a shallow-cloned event with all free-text fields scrubbed of secrets.
 */
export function sanitizeEvent<T extends CaptureEvent>(event: T): T {
  const clone: any = { ...event };
  if (typeof clone.command === "string") clone.command = sanitizeText(clone.command);
  if (Array.isArray(clone.args)) {
    clone.args = clone.args.map((a: any) => (typeof a === "string" ? sanitizeText(a) : a));
  }
  if (typeof clone.stdout === "string") clone.stdout = sanitizeText(clone.stdout);
  if (typeof clone.stderr === "string") clone.stderr = sanitizeText(clone.stderr);
  if (typeof clone.diff === "string") clone.diff = sanitizeText(clone.diff);
  if (typeof clone.staged_diff === "string") clone.staged_diff = sanitizeText(clone.staged_diff);
  return clone as T;
}
```

```ts
import { existsSync, renameSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import type { CaptureEvent, GitSnapshotEvent, ToolCallEvent } from "./events.js";
import { EventLog } from "./log.js";
import { isExtractionComplete } from "../extraction-lock.js";
import { getEventsLogPath } from "../local-data.js";

const PASS_STDOUT_LIMIT = 300;
const FAIL_OUTPUT_LIMIT = 1500;
const DIFF_LIMIT = 2000;

function truncate(text: string | undefined, max: number): string | undefined {
  if (!text) return text;
  return text.length > max ? text.slice(0, max) + "...[truncated]" : text;
}

/**
 * Deterministically compacts raw session events before building the LLM extraction
 * prompt, reducing token usage by 70-90% while preserving failure diagnostics.
 */
export function compactSessionEventsForPrompt(events: CaptureEvent[]): CaptureEvent[] {
  const result: CaptureEvent[] = [];
  const seenReads = new Set<string>();

  // Identify the index of the last git_snapshot so earlier snapshots can drop bulky diffs
  let lastGitSnapIdx = -1;
  for (let i = events.length - 1; i >= 0; i--) {
    if (events[i].type === "git_snapshot") {
      lastGitSnapIdx = i;
      break;
    }
  }

  for (let i = 0; i < events.length; i++) {
    const ev = events[i];

    if (ev.type === "tool_call") {
      const tc = ev as ToolCallEvent;
      // Drop pre-execution sentinel events (exit_code === -1)
      if (tc.exit_code === -1) continue;

      // Deduplicate repeated Read calls on the same file
      if (tc.command.toLowerCase() === "read" && tc.args.length > 0) {
        const readKey = tc.args[0];
        if (seenReads.has(readKey)) continue;
        seenReads.add(readKey);
      }

      const isFailure = tc.exit_code !== 0;
      const limit = isFailure ? FAIL_OUTPUT_LIMIT : PASS_STDOUT_LIMIT;
      result.push({
        ...tc,
        stdout: truncate(tc.stdout, limit),
        stderr: truncate(tc.stderr, FAIL_OUTPUT_LIMIT),
      });
      continue;
    }

    if (ev.type === "git_snapshot") {
      const gs = ev as GitSnapshotEvent;
      if (i !== lastGitSnapIdx) {
        result.push({ ...gs, diff: "", staged_diff: "" });
      } else {
        result.push({
          ...gs,
          diff: truncate(gs.diff, DIFF_LIMIT) ?? "",
          staged_diff: truncate(gs.staged_diff, DIFF_LIMIT) ?? "",
        });
      }
      continue;
    }

    result.push(ev);
  }

  return result;
}

/**
 * Atomically prunes events belonging to sessions that have already been
 * extracted into the SQLite graph store.
 */
export function pruneExtractedEvents(projectRoot: string): { removedEvents: number; keptEvents: number } {
  const logPath = getEventsLogPath(projectRoot);
  if (!existsSync(logPath)) {
    return { removedEvents: 0, keptEvents: 0 };
  }

  const log = new EventLog({ persistPath: logPath });
  const allEvents = log.getEvents();
  const kept: CaptureEvent[] = [];
  let removedEvents = 0;

  const completionCache = new Map<string, boolean>();
  for (const ev of allEvents) {
    let done = completionCache.get(ev.session_id);
    if (done === undefined) {
      done = isExtractionComplete(projectRoot, ev.session_id);
      completionCache.set(ev.session_id, done);
    }
    if (done) {
      removedEvents++;
    } else {
      kept.push(ev);
    }
  }

  if (removedEvents > 0) {
    const tmpPath = join(projectRoot, ".dev-mem", `events.jsonl.tmp.${process.pid}`);
    const body = kept.length > 0 ? kept.map((e) => JSON.stringify(e)).join("\n") + "\n" : "";
    writeFileSync(tmpPath, body, "utf8");
    renameSync(tmpPath, logPath);
  }

  return { removedEvents, keptEvents: kept.length };
}
```

```ts
import { existsSync, mkdirSync, readFileSync, writeFileSync } from "node:fs";
import { join } from "node:path";

function seenFilePath(projectRoot: string, sessionId: string): string {
  const dir = join(projectRoot, ".dev-mem", "sessions");
  mkdirSync(dir, { recursive: true });
  return join(dir, `${encodeURIComponent(sessionId)}.injected.json`);
}

/**
 * Filters a candidate list of nodes down to those not yet injected during
 * `sessionId`, and records the newly selected node IDs as seen.
 */
export function filterUnseenNodes<T extends { id: string }>(
  projectRoot: string,
  sessionId: string,
  nodes: T[],
): T[] {
  if (!sessionId || nodes.length === 0) return nodes;
  try {
    const path = seenFilePath(projectRoot, sessionId);
    let seenIds: string[] = [];
    if (existsSync(path)) {
      try {
        const parsed = JSON.parse(readFileSync(path, "utf8"));
        if (Array.isArray(parsed)) seenIds = parsed;
      } catch {
        seenIds = [];
      }
    }
    const seenSet = new Set(seenIds);
    const unseen = nodes.filter((n) => !seenSet.has(n.id));
    if (unseen.length > 0) {
      for (const n of unseen) seenSet.add(n.id);
      writeFileSync(path, JSON.stringify([...seenSet]), "utf8");
    }
    return unseen;
  } catch {
    return nodes;
  }
}
```

```ts
export async function handleMcpRequest(request: any, projectRoot: string): Promise<any | null> {
  if (!request || typeof request !== "object") return null;
  const { id, method, params } = request;

  // Notifications have no id and expect no response
  if (id === undefined || id === null) return null;

  try {
    if (method === "initialize") {
      return {
        jsonrpc: "2.0",
        id,
        result: {
          protocolVersion: "2024-11-05",
          capabilities: { tools: {} },
          serverInfo: { name: "dev-memo", version: "0.4.0" },
        },
      };
    }

    if (method === "ping") {
      return { jsonrpc: "2.0", id, result: {} };
    }

    if (method === "tools/list") {
      return { jsonrpc: "2.0", id, result: { tools: MCP_TOOLS } };
    }

    if (method === "tools/call") {
      const toolName = params?.name;
      const args = params?.arguments ?? {};

      if (toolName === "dev_mem_query") {
        const query = typeof args.query === "string" ? args.query : "";
        const files = Array.isArray(args.files) ? args.files.map(String) : undefined;
        const nodes = retrieveContext({ projectRoot, currentTask: query, currentFiles: files, budgetTokens: 1500 });
        const text = nodes.length > 0
          ? nodes.map((n) => `[${n.type}] (id: ${n.id}, state: ${n.lifecycle_state}, score: ${n.score.toFixed(2)})\nTitle: ${n.title}\nDetail: ${n.content}\nFiles: ${n.evidence.files.join(", ")}`).join("\n\n")
          : "No matching knowledge nodes found.";
        return { jsonrpc: "2.0", id, result: { content: [{ type: "text", text }] } };
      }

      if (toolName === "dev_mem_check_regression") {
        const files = Array.isArray(args.files) ? args.files.map(String) : [];
        const matches = checkRegressionRisk({ projectRoot, currentFiles: files, confidenceThreshold: 0.5 });
        const text = matches.length > 0 ? formatRegressionWarning(matches) : "No known regression risks for these files.";
        return { jsonrpc: "2.0", id, result: { content: [{ type: "text", text }] } };
      }

      if (toolName === "dev_mem_remember") {
        const type = args.type as NodeType;
        const title = sanitizeText(String(args.title ?? "").trim());
        const content = sanitizeText(String(args.content ?? "").trim());
        const files = Array.isArray(args.files) && args.files.length > 0 ? args.files.map(String) : ["README.md"];
        const symbols = Array.isArray(args.symbols) ? args.symbols.map(String) : [];

        let commit = "0000000000000000000000000000000000000000";
        try {
          commit = collectGitSnapshot(projectRoot).head_commit || commit;
        } catch {}

        const store = new GraphStore({ projectRoot });
        try {
          const node = store.createNode({
            type,
            title,
            content,
            lifecycle_state: "active",
            evidence: {
              commit,
              files,
              symbols,
              session_id: "mcp_direct",
              agent: "mcp",
              timestamp: new Date().toISOString(),
              confidence: 0.95,
            },
          });
          return {
            jsonrpc: "2.0",
            id,
            result: { content: [{ type: "text", text: `Recorded [${node.type}] (id: ${node.id}): ${node.title}` }] },
          };
        } finally {
          store.close();
        }
      }

      if (toolName === "dev_mem_resolve") {
        const nodeId = String(args.node_id ?? "");
        const toState = args.to_state as any;
        const store = new GraphStore({ projectRoot });
        try {
          const updated = store.transitionLifecycleState(nodeId, toState);
          return {
            jsonrpc: "2.0",
            id,
            result: { content: [{ type: "text", text: `Transitioned node ${updated.id} ("${updated.title}") to ${updated.lifecycle_state}.` }] },
          };
        } finally {
          store.close();
        }
      }

      return {
        jsonrpc: "2.0",
        id,
        result: { isError: true, content: [{ type: "text", text: `Unknown tool: ${toolName}` }] },
      };
    }

    return {
      jsonrpc: "2.0",
      id,
      error: { code: -32601, message: `Method not found: ${method}` },
    };
  } catch (err: any) {
    return {
      jsonrpc: "2.0",
      id,
      result: { isError: true, content: [{ type: "text", text: `Error: ${err.message}` }] },
    };
  }
}

export function startMcpStdioServer(projectRoot: string): Promise<void> {
  return new Promise((resolve) => {
    const rl = createInterface({ input: process.stdin, terminal: false });
    rl.on("line", async (line) => {
      const trimmed = line.trim();
      if (!trimmed) return;
      try {
        const req = JSON.parse(trimmed);
        const res = await handleMcpRequest(req, projectRoot);
        if (res) {
          process.stdout.write(JSON.stringify(res) + "\n");
        }
      } catch {
        // Ignore malformed JSON lines on stdin
      }
    });
    rl.on("close", () => resolve());
  });
}
```

```ts
import { existsSync, readFileSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import { GraphStore, type KnowledgeEdge, type KnowledgeNode } from "../graph/index.js";
import { ensureLocalDataDir } from "../local-data.js";

export const SHARED_GRAPH_FILENAME = "shared.jsonl";

export function getSharedGraphPath(projectRoot: string): string {
  return join(projectRoot, ".dev-mem", SHARED_GRAPH_FILENAME);
}

interface SharedNodeRecord {
  record_type: "node";
  node: KnowledgeNode;
}

interface SharedEdgeRecord {
  record_type: "edge";
  edge: KnowledgeEdge;
}

type SharedRecord = SharedNodeRecord | SharedEdgeRecord;

/**
 * Exports all curated (`active` and `verified`) knowledge nodes and their
 * connecting edges to `.dev-mem/shared.jsonl`, sorted deterministically by ID
 * so git diffs and merges are clean.
 */
export function exportSharedGraph(projectRoot: string): {
  exportedNodes: number;
  exportedEdges: number;
  path: string;
} {
  ensureLocalDataDir(projectRoot);
  const store = new GraphStore({ projectRoot });
  try {
    const shareableNodes = store
      .queryAllNodes()
      .filter((n) => n.lifecycle_state === "active" || n.lifecycle_state === "verified")
      .sort((a, b) => a.id.localeCompare(b.id));

    const nodeIdSet = new Set(shareableNodes.map((n) => n.id));
    const shareableEdges = store
      .listEdges()
      .filter((e) => nodeIdSet.has(e.from_id) && nodeIdSet.has(e.to_id))
      .sort((a, b) => a.id.localeCompare(b.id));

    const lines: string[] = [
      ...shareableNodes.map((node) => JSON.stringify({ record_type: "node", node } satisfies SharedNodeRecord)),
      ...shareableEdges.map((edge) => JSON.stringify({ record_type: "edge", edge } satisfies SharedEdgeRecord)),
    ];

    const outPath = getSharedGraphPath(projectRoot);
    writeFileSync(outPath, lines.length > 0 ? lines.join("\n") + "\n" : "", "utf8");
    return {
      exportedNodes: shareableNodes.length,
      exportedEdges: shareableEdges.length,
      path: outPath,
    };
  } finally {
    store.close();
  }
}

/**
 * Idempotently imports nodes and edges from `.dev-mem/shared.jsonl` into the
 * local SQLite graph store. Safe to call on SessionStart or via `dev-memo sync`.
 */
export function importSharedGraph(projectRoot: string): {
  importedNodes: number;
  updatedNodes: number;
  importedEdges: number;
} {
  const sharedPath = getSharedGraphPath(projectRoot);
  if (!existsSync(sharedPath)) {
    return { importedNodes: 0, updatedNodes: 0, importedEdges: 0 };
  }

  const raw = readFileSync(sharedPath, "utf8").trim();
  if (!raw) {
    return { importedNodes: 0, updatedNodes: 0, importedEdges: 0 };
  }

  let importedNodes = 0;
  let updatedNodes = 0;
  let importedEdges = 0;

  const store = new GraphStore({ projectRoot });
  try {
    const existingEdges = new Set(store.listEdges().map((e) => e.id));

    for (const line of raw.split(/\r?\n/)) {
      if (!line.trim()) continue;
      let rec: SharedRecord;
      try {
        rec = JSON.parse(line) as SharedRecord;
      } catch {
        continue;
      }

      if (rec.record_type === "node" && rec.node?.id) {
        const existing = store.getNode(rec.node.id);
        if (!existing) {
          try {
            store.createNode({
              id: rec.node.id,
              type: rec.node.type,
              title: rec.node.title,
              content: rec.node.content,
              lifecycle_state: rec.node.lifecycle_state,
              evidence: rec.node.evidence,
            });
            importedNodes++;
          } catch {
            // Skip invalid node
          }
        } else if (
          existing.lifecycle_state === "observed" &&
          (rec.node.lifecycle_state === "active" || rec.node.lifecycle_state === "verified")
        ) {
          try {
            store.transitionLifecycleState(existing.id, rec.node.lifecycle_state);
            updatedNodes++;
          } catch {}
        }
      } else if (rec.record_type === "edge" && rec.edge?.id) {
        if (!existingEdges.has(rec.edge.id)) {
          try {
            store.createEdge({
              id: rec.edge.id,
              from_id: rec.edge.from_id,
              to_id: rec.edge.to_id,
              type: rec.edge.type,
            });
            existingEdges.add(rec.edge.id);
            importedEdges++;
          } catch {}
        }
      }
    }
  } finally {
    store.close();
  }

  return { importedNodes, updatedNodes, importedEdges };
}

/**
 * Bidirectional sync: first imports any teammate updates from `.dev-mem/shared.jsonl`,
 * then exports all local `active` and `verified` nodes back to `.dev-mem/shared.jsonl`.
 */
export function syncSharedGraph(projectRoot: string): {
  importedNodes: number;
  updatedNodes: number;
  exportedNodes: number;
} {
  const imp = importSharedGraph(projectRoot);
  const exp = exportSharedGraph(projectRoot);
  return {
    importedNodes: imp.importedNodes,
    updatedNodes: imp.updatedNodes,
    exportedNodes: exp.exportedNodes,
  };
}
```
